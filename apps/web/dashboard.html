JS SDK
npm install @forsig/sdk

import { createForsigOpenAIClient, forsigHeaders } from "@forsig/sdk";

const client = createForsigOpenAIClient({
  apiKey: process.env.FORSIG_API_KEY,
});

await client.chat.completions.create({
  model: "gpt-4o-mini",
  messages: [{ role: "user", content: "Hello" }],
}, {
  headers: forsigHeaders({ externalUserId: "user_123" }),
});
Python SDK
pip install forsig

from forsig import create_forsig_openai_client, forsig_headers

client = create_forsig_openai_client(api_key=os.environ["FORSIG_API_KEY"])

client.chat.completions.create(
    model="gpt-4o-mini",
    messages=[{"role": "user", "content": "Hello"}],
    extra_headers=forsig_headers(external_user_id="user_123"),
)
