Run a real approval in minutes.

Forsig pauses a risky AI-agent action, asks a human for a decision, and returns a structured result your code can branch on.

Quickstart

  1. Create or use the starter agent in Developer.
  2. Create an API key. Forsig shows the full key once.
  3. Set FORSIG_API_KEY locally.
  4. Run one of the examples below and approve the Inbox escalation.
export FORSIG_API_KEY=fsk_test_xxx
export FORSIG_BASE_URL=https://www.forsig.com
npx create-forsig-demo

Plain Node

const baseUrl = process.env.FORSIG_BASE_URL || "https://www.forsig.com";
const apiKey = process.env.FORSIG_API_KEY;

const payload = {
  agent: { id: "refund-agent", name: "Refund Agent", environment: "production" },
  workflow: "refund-review-flow",
  step: "refund-over-limit",
  risk: { type: "refund_over_limit", level: "high" },
  task: {
    title: "Approve refund for customer #123",
    proposedAction: "Issue a $500 refund",
    customerImpact: true
  },
  context: { refundAmount: 500, customerTier: "VIP" },
  review: { notify: ["dashboard", "email"] }
};

const created = await fetch(`${baseUrl}/api/v1/escalations`, {
  method: "POST",
  headers: {
    "authorization": `Bearer ${apiKey}`,
    "content-type": "application/json"
  },
  body: JSON.stringify(payload)
}).then((res) => res.json());

console.log("Open in Forsig:", created.dashboardUrl || created.dashboard_url);

No secrets: send the smallest review context possible. Do not include API keys, passwords, full payment details, or unnecessary personal data.

Python

import os, requests, time

base_url = os.getenv("FORSIG_BASE_URL", "https://www.forsig.com")
headers = {
    "authorization": f"Bearer {os.environ['FORSIG_API_KEY']}",
    "content-type": "application/json",
}

payload = {
    "agent": {"id": "refund-agent", "name": "Refund Agent", "environment": "production"},
    "risk": {"type": "refund_over_limit", "level": "high"},
    "task": {"title": "Approve refund for customer #123", "proposedAction": "Issue a $500 refund"},
    "context": {"refundAmount": 500, "customerTier": "VIP"},
    "review": {"notify": ["dashboard", "email"]},
}

created = requests.post(f"{base_url}/api/v1/escalations", json=payload, headers=headers).json()
while True:
    time.sleep(3)
    result = requests.get(f"{base_url}/api/v1/escalations/{created['id']}", headers=headers).json()
    if result["escalation"]["status"] != "pending":
        print(result["escalation"]["decision"])
        break

Framework examples

LangGraph

async function refundApprovalNode(state) {
  const decision = await forsig.escalate({
    agent: { id: "refund-agent", name: "Refund Agent" },
    risk: { type: "refund_over_limit", level: "high" },
    task: { title: "Approve refund", proposedAction: "Issue refund" },
    context: { refundAmount: state.refundAmount },
    waitForDecision: true
  });
  return decision.status === "approved" ? { ...state, approved: true } : { ...state, stopped: true };
}

Webhook signing

When callbackUrl is set, Forsig posts the resolved decision with x-forsig-event and x-forsig-signature. Verify the raw request body before trusting the decision.

import { createHmac, timingSafeEqual } from "node:crypto";

function verifyForsigWebhook(rawBody, signature, secret) {
  const expected = createHmac("sha256", secret).update(rawBody).digest("hex");
  const a = Buffer.from(signature || "");
  const b = Buffer.from(expected);
  return a.length === b.length && timingSafeEqual(a, b);
}
